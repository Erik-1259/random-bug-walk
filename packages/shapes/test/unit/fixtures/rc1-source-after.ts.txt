import prisma from '@/lib/synthetic-prisma';

export async function POST(request: Request) {
  const { userId, backupCode } = await request.json();
  const unusedCodes = await prisma.client.twoFactorBackupCode.findMany({
    where: { userId, used: false },
  });
  const matchIndex = await verifySyntheticCode(backupCode, unusedCodes);

  const consumed = await prisma.client.twoFactorBackupCode.updateMany({
    where: { id: unusedCodes[matchIndex].id, used: false },
    data: { used: true },
  });

  if (consumed.count === 0) {
    return badRequest({ code: 'synthetic-invalid-code' });
  }
  return json({ ok: true });
}
