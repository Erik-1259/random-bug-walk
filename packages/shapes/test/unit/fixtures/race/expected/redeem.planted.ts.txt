import { db } from '@/lib/db';
import { verifyVoucherCode } from '@/lib/vouchers';

export async function POST(request: Request) {
  const { ownerId, code } = await request.json();

  const vouchers = await db.client.voucher.findMany({
    where: { ownerId, redeemed: false },
  });

  const index = await verifyVoucherCode(vouchers, code);
  if (index < 0) {
    return Response.json({ error: 'invalid' }, { status: 400 });
  }

  const spent = await db.client.voucher.updateMany({
    where: { id: vouchers[index].id },
    data: { redeemed: true },
  });

  if (spent.count !== 1) {
    return Response.json({ error: 'already redeemed' }, { status: 409 });
  }

  await db.client.ledger.updateMany({
    where: { ownerId, settled: false },
    data: { touchedAt: new Date() },
  });

  return Response.json({ ok: true });
}
